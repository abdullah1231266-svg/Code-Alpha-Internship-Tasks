<?php
// SECURE LOGIN SYSTEM - PRODUCTION READY

// Secure configuration
session_start([
    'cookie_httponly' => true,
    'cookie_secure' => true,
    'cookie_samesite' => 'Strict',
    'use_strict_mode' => true
]);

// Regenerate session ID to prevent fixation
session_regenerate_id(true);

// Database connection with PDO (prepared statements)
class Database {
    private $pdo;

    public function __construct() {
        try {
            $this->pdo = new PDO(
                "mysql:host=localhost;dbname=testdb;charset=utf8mb4",
                "root",
                "",
                [
                    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                    PDO::ATTR_EMULATE_PREPARES => false
                ]
            );
        } catch(PDOException $e) {
            // Log error without exposing details
            error_log("Database connection failed: " . $e->getMessage());
            die("System error. Please try again later.");
        }
    }

    public function getConnection() {
        return $this->pdo;
    }
}

// Input validation class
class Validator {
    public static function sanitizeInput($data) {
        $data = trim($data);
        $data = stripslashes($data);
        $data = htmlspecialchars($data, ENT_QUOTES, 'UTF-8');
        return $data;
    }

    public static function validateEmail($email) {
        return filter_var($email, FILTER_VALIDATE_EMAIL);
    }

    public static function validateUsername($username) {
        return preg_match('/^[a-zA-Z0-9_]{3,20}$/', $username);
    }

    public static function validateIP($ip) {
        return filter_var($ip, FILTER_VALIDATE_IP);
    }
}

// Rate limiting class
class RateLimiter {
    private $maxAttempts = 5;
    private $timeWindow = 900; // 15 minutes

    public function checkAttempts($ip) {
        $attempts = $_SESSION['login_attempts'][$ip] ?? 0;
        $firstAttempt = $_SESSION['login_first_attempt'][$ip] ?? time();

        if($attempts >= $this->maxAttempts && (time() - $firstAttempt) < $this->timeWindow) {
            return false;
        }

        if(time() - $firstAttempt > $this->timeWindow) {
            $_SESSION['login_attempts'][$ip] = 0;
            $_SESSION['login_first_attempt'][$ip] = time();
        }

        return true;
    }

    public function incrementAttempts($ip) {
        $_SESSION['login_attempts'][$ip] = ($_SESSION['login_attempts'][$ip] ?? 0) + 1;
    }

    public function resetAttempts($ip) {
        $_SESSION['login_attempts'][$ip] = 0;
    }
}

// Password hashing class
class PasswordHandler {
    public static function hashPassword($password) {
        return password_hash($password, PASSWORD_ARGON2ID, [
            'memory_cost' => 65536,
            'time_cost' => 4,
            'threads' => 3
        ]);
    }

    public static function verifyPassword($password, $hash) {
        return password_verify($password, $hash);
    }
}

// CSRF protection
class CSRFToken {
    public static function generate() {
        if(empty($_SESSION['csrf_token'])) {
            $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
        }
        return $_SESSION['csrf_token'];
    }

    public static function validate($token) {
        if(!isset($_SESSION['csrf_token']) || !hash_equals($_SESSION['csrf_token'], $token)) {
            throw new Exception("CSRF token validation failed");
        }
        return true;
    }
}

// Main login handler
class LoginHandler {
    private $db;
    private $rateLimiter;

    public function __construct($db, $rateLimiter) {
        $this->db = $db;
        $this->rateLimiter = $rateLimiter;
    }

    public function processLogin($username, $password, $csrfToken) {
        try {
            // Validate CSRF token
            CSRFToken::validate($csrfToken);

            // Get client IP
            $clientIP = $_SERVER['REMOTE_ADDR'] ?? '0.0.0.0';

            // Check rate limiting
            if(!$this->rateLimiter->checkAttempts($clientIP)) {
                error_log("Rate limit exceeded for IP: " . $clientIP);
                return ["success" => false, "message" => "Too many attempts. Try again later."];
            }

            // Validate input
            if(!Validator::validateUsername($username)) {
                return ["success" => false, "message" => "Invalid username format"];
            }

            // Use prepared statement
            $stmt = $this->db->prepare("SELECT id, username, password_hash, failed_attempts FROM users WHERE username = ?");
            $stmt->execute([$username]);
            $user = $stmt->fetch();

            if(!$user) {
                $this->rateLimiter->incrementAttempts($clientIP);
                // Generic message to prevent user enumeration
                return ["success" => false, "message" => "Invalid credentials"];
            }

            // Verify password
            if(!PasswordHandler::verifyPassword($password, $user['password_hash'])) {
                $this->rateLimiter->incrementAttempts($clientIP);
                return ["success" => false, "message" => "Invalid credentials"];
            }

            // Reset rate limiting on success
            $this->rateLimiter->resetAttempts($clientIP);

            // Regenerate session ID
            session_regenerate_id(true);

            // Set secure session variables
            $_SESSION['user_id'] = $user['id'];
            $_SESSION['username'] = $user['username'];
            $_SESSION['login_time'] = time();
            $_SESSION['ip_address'] = $clientIP;

            // Log successful login
            error_log("Successful login for user: " . $username . " from IP: " . $clientIP);

            return ["success" => true, "message" => "Login successful", "redirect" => "/dashboard"];

        } catch(Exception $e) {
            error_log("Login error: " . $e->getMessage());
            return ["success" => false, "message" => "System error. Please try again."];
        }
    }
}

// Only process POST requests
if($_SERVER['REQUEST_METHOD'] === 'POST') {
    $db = new Database();
    $rateLimiter = new RateLimiter();
    $loginHandler = new LoginHandler($db->getConnection(), $rateLimiter);

    $result = $loginHandler->processLogin(
        $_POST['username'] ?? '',
        $_POST['password'] ?? '',
        $_POST['csrf_token'] ?? ''
    );

    header('Content-Type: application/json');
    echo json_encode($result);
    exit;
}
